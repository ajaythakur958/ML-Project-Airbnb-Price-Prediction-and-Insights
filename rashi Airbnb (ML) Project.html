def missing_values_table(df):
    mis_val = df.isnull().sum()
    mis_val_percent = 100 * df.isnull().sum() / len(df)
    mis_val_table = pd.concat([mis_val, mis_val_percent], axis=1)
    mis_val_table = mis_val_table.rename(columns={0: 'Missing Values', 1: '% of Total Values'})
    mis_val_table = mis_val_table[mis_val_table['Missing Values'] > 0].sort_values('% of Total Values', ascending=False)
    return mis_val_table

#  Function to Detect Outliers in Numeric Columns
def detect_outliers(df):
    outlier_summary = {}
    numeric_columns = df.select_dtypes(include=[np.number]).columns
    
    for column in numeric_columns:
        Q1 = df[column].quantile(0.25)
        Q3 = df[column].quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
        outliers = df[(df[column] < lower_bound) | (df[column] > upper_bound)]
        
        outliers_percentage = round((len(outliers) / len(df)) * 100, 2)
        
        outlier_summary[column] = {
            "Total Values": len(df),
            "Outliers": len(outliers),
            "Outlier %": outliers_percentage
        }
    
    # Return a summary table sorted by the highest outlier percentage
    return pd.DataFrame(outlier_summary).T.sort_values(by="Outlier %", ascending=False)

Outlier Detection & Handling¶

1) df['bedrooms'] > 10 → 10: Any property with more than 10 bedrooms is capped at 10 (to handle unrealistic listings).¶

2) df['bathrooms'] > 8 → 8: Any listing with more than 8 bathrooms is capped at 8.¶

3) For columns ['number_of_reviews', 'beds', 'host_response_rate']:¶

The function caps values above the 99th percentile to prevent rare, extreme high values from skewing the dataset.¶

4) For columns ['log_price', 'accommodates', 'review_scores_rating']:¶

Values below the 1st percentile and above the 99th percentile are clipped within that range using np.clip() to smooth out both lower and upper extremes.¶

In [15]:
def handle_outliers(df):
    df['bedrooms'] = np.where(df['bedrooms'] > 10, 10, df['bedrooms'])
    df['bathrooms'] = np.where(df['bathrooms'] > 8, 8, df['bathrooms'])
    for col in ['number_of_reviews', 'beds', 'host_response_rate']:
        upper_limit = df[col].quantile(0.99)
        df[col] = np.where(df[col] > upper_limit, upper_limit, df[col])
    for col in ['log_price', 'accommodates', 'review_scores_rating']:
        lower_limit = df[col].quantile(0.01)
        upper_limit = df[col].quantile(0.99)
        df[col] = np.clip(df[col], lower_limit, upper_limit)
    return df

def impute_missing_values(df):
    for col in df.columns:

            df[col].fillna(df[col].median(), inplace=True)
        else:
